# ---------------------------------------------------------------------------
# Blackwater.Core -- the engine, as a STATIC library.
#
# Static, not shared: a DLL boundary is an ABI boundary and C++ has no stable
# ABI. Linking statically lets STL types, templates and exceptions cross freely
# between engine and game. When the WPF editor eventually needs to call in, it
# gets a separate, deliberately tiny extern "C" DLL (Blackwater.Interop).
# ---------------------------------------------------------------------------

# Sources are listed EXPLICITLY, never via file(GLOB).
#
# GLOB is evaluated at *configure* time, so a newly added .cpp would silently
# not be built until someone regenerated by hand. (This is the opposite of an
# SDK-style .csproj, which globs **/*.cs for you.) We extend this list by hand
# as each file arrives.
#
# Headers are listed too. They are not compiled -- they are here so they show
# up in Solution Explorer instead of being invisible.
set(BLACKWATER_CORE_SOURCES
    include/Blackwater/Application.h
    include/Blackwater/Check.h
    include/Blackwater/ConstantBuffer.h
    include/Blackwater/GraphicsDevice.h
    include/Blackwater/Mesh.h
    include/Blackwater/Shader.h
    include/Blackwater/Window.h
    src/Application.cpp
    src/Check.cpp
    src/GraphicsDevice.cpp
    src/Mesh.cpp
    src/Shader.cpp
    src/Window.cpp
)

add_library(Blackwater.Core STATIC ${BLACKWATER_CORE_SOURCES})

# ---------------------------------------------------------------------------
# Include paths
#
# PUBLIC  include/ -> used to build Core AND propagated to anything linking it,
#                     so consumers write  #include <Blackwater/Check.h>
# PRIVATE src/     -> internal headers; unreachable from outside the engine.
#                     The public API surface is therefore physical, not a
#                     convention someone has to remember.
# ---------------------------------------------------------------------------
target_include_directories(Blackwater.Core
    PUBLIC  "${CMAKE_CURRENT_SOURCE_DIR}/include"
    PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src"
)

# ---------------------------------------------------------------------------
# Dependencies
#
# Blackwater.Common is PUBLIC: the C++20 dialect, /W4 and the Win32 defines
# must apply to Sandbox and Trinity too, or they would compile against these
# headers with different settings.
#
# The D3D libraries are PRIVATE: they are implementation detail, not part of
# Core's API contract. A static library never links, so CMake still forwards
# these to the final executable as $<LINK_ONLY:...> -- Sandbox.exe gets
# d3d11.lib without inheriting anything else.
#
# ---------------------------------------------------------------------------
target_link_libraries(Blackwater.Core
    PUBLIC
        Blackwater.Common
    PRIVATE
        d3d11         # device, immediate context, resources
        dxgi          # swap chain, adapters, presentation
        d3dcompiler   # runtime HLSL compilation (M1)
)

# Mirror the on-disk folder structure in the IDE rather than dumping every
# file into one flat "Source Files" node.
source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${BLACKWATER_CORE_SOURCES})

set_target_properties(Blackwater.Core PROPERTIES FOLDER "Engine")
